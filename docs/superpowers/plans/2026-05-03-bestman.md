```python
"""SQLite state management for bestman. Persists daily progress, streak,
skip tokens, voyage logs, and milestones. Uses WAL mode for concurrency."""

import datetime
import json
import sqlite3
from pathlib import Path
from contextlib import contextmanager


SCHEMA = """
CREATE TABLE IF NOT EXISTS days (
    date TEXT PRIMARY KEY,
    completed INTEGER NOT NULL DEFAULT 0,
    extra INTEGER NOT NULL DEFAULT 0,
    task_done TEXT DEFAULT '',
    used_skip INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS voyage_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    date TEXT NOT NULL,
    text TEXT NOT NULL,
    event_type TEXT DEFAULT 'narrative',
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS milestones (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    date TEXT NOT NULL,
    description TEXT DEFAULT '',
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS skip_tokens (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    earned_on_date TEXT,
    used INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE INDEX IF NOT EXISTS idx_days_date ON days(date);
CREATE INDEX IF NOT EXISTS idx_logs_date ON voyage_logs(date);
"""


class BestmanState:
    def __init__(self, db_path=None):
        if db_path is None:
            db_path = Path.home() / ".bestman" / "state.db"
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._conn = sqlite3.connect(str(self.db_path), check_same_thread=False)
        self._conn.execute("PRAGMA journal_mode=WAL")
        self._conn.row_factory = sqlite3.Row
        self._cursor = self._conn.cursor()
        self._cursor.executescript(SCHEMA)
        self._conn.commit()

    def record_day(self, date, completed, extra=0, task_done="", used_skip=False):
        self._cursor.execute(
            """INSERT OR REPLACE INTO days (date, completed, extra, task_done, used_skip)
               VALUES (?, ?, ?, ?, ?)""",
            (date, int(completed), extra, task_done, int(used_skip)),
        )
        # Earn skip token every 7-day streak
        if completed and self.get_streak() == 7:
            self.add_skip_token()
        self._conn.commit()

    def get_progress(self):
        self._cursor.execute("SELECT COUNT(*) as c FROM days WHERE completed = 1")
        total = self._cursor.fetchone()["c"]
        self._cursor.execute("SELECT SUM(extra) as e FROM days WHERE completed = 1")
        extra = self._cursor.fetchone()["e"] or 0
        return {
            "total_completed": total + (extra or 0),
            "streak": self.get_streak(),
            "skip_tokens": self.get_skip_tokens(),
        }

    def get_streak(self):
        today = datetime.date.today().isoformat()
        self._cursor.execute(
            "SELECT date, completed FROM days WHERE date <= ? ORDER BY date DESC LIMIT 200",
            (today,),
        )
        rows = self._cursor.fetchall()
        streak = 0
        expected = datetime.date.today()
        for row in rows:
            row_date = datetime.date.fromisoformat(row["date"])
            if row_date == expected and row["completed"]:
                streak += 1
                expected = expected - datetime.timedelta(days=1)
            elif row_date < expected:
                break
            elif row_date == expected and not row["completed"]:
                break
        return streak

    def add_skip_token(self):
        today = datetime.date.today().isoformat()
        self._cursor.execute(
            "INSERT INTO skip_tokens (earned_on_date, used) VALUES (?, 0)", (today,)
        )
        self._conn.commit()

    def get_skip_tokens(self):
        self._cursor.execute("SELECT COUNT(*) as c FROM skip_tokens WHERE used = 0")
        return self._cursor.fetchone()["c"]

    def use_skip_token(self):
        self._cursor.execute(
            "SELECT id FROM skip_tokens WHERE used = 0 ORDER BY id LIMIT 1"
        )
        row = self._cursor.fetchone()
        if row is None:
            return False
        self._cursor.execute("UPDATE skip_tokens SET used = 1 WHERE id = ?", (row["id"],))
        self._conn.commit()
        return True

    def save_voyage_log(self, date, text, event_type="narrative"):
        self._cursor.execute(
            "INSERT INTO voyage_logs (date, text, event_type) VALUES (?, ?, ?)",
            (date, text, event_type),
        )
        self._conn.commit()

    def get_voyage_logs(self, limit=10):
        self._cursor.execute(
            "SELECT * FROM voyage_logs ORDER BY date DESC LIMIT ?", (limit,)
        )
        return [dict(row) for row in self._cursor.fetchall()]

    def mark_milestone(self, name, date, description=""):
        self._cursor.execute(
            "INSERT INTO milestones (name, date, description) VALUES (?, ?, ?)",
            (name, date, description),
        )
        self._conn.commit()

    def get_milestones(self):
        self._cursor.execute("SELECT * FROM milestones ORDER BY date")
        return [dict(row) for row in self._cursor.fetchall()]

    def get_current_stage_info(self, start_date="2026-05-03", total_days=175):
        self._cursor.execute("SELECT COUNT(*) as c FROM days WHERE completed = 1")
        completed = self._cursor.fetchone()["c"]
        self._cursor.execute("SELECT SUM(extra) as e FROM days WHERE completed = 1")
        extra = self._cursor.fetchone()["e"] or 0
        current_day = completed + (extra or 0) + 1  # tomorrow = current + 1
        return {
            "current_day": min(current_day, total_days),
            "total_days": total_days,
            "days_remaining": max(total_days - current_day + 1, 0),
        }

    def close(self):
        self._conn.close()
```

```python
"""Abstract base for bestman themes. Each theme defines the visual vocabulary
for the pixel map: tile characters, colors, and special markers."""

from abc import ABC, abstractmethod
from dataclasses import dataclass, field


@dataclass
class ThemeTileSet:
    """Characters and colors for each map tile type."""
    unexplored: str = "░"
    explored_empty: str = "·"
    ship: str = "⚓"
    milestone: str = "✦"
    landmark: str = "◆"
    festival: str = "★"
    dock: str = "⬡"
    fog: str = "▒"
    # Rich style strings
    unexplored_style: str = "dim cyan"
    explored_style: str = "green"
    ship_style: str = "bold yellow"
    milestone_style: str = "bold magenta"
    landmark_style: str = "bold cyan"
    festival_style: str = "bold yellow blink"
    dock_style: str = "bright_blue"


class BaseTheme(ABC):
    name: str = "base"
    display_name: str = "Base Theme"
    tile_set: ThemeTileSet = field(default_factory=ThemeTileSet)

    @abstractmethod
    def get_map_dimensions(self, total_days):
        """Return (width, height) for the map grid."""
        ...

    @abstractmethod
    def get_milestone_names(self):
        """Return dict of {day_number: milestone_name} for key landmarks."""
        ...

    @abstractmethod
    def get_narrative_tone(self):
        """Return a string describing the narrative voice for LLM prompts."""
        ...
```

```python
"""Pixel-art fog-of-war map engine for bestman."""

from dataclasses import dataclass, field
from typing import Optional


@dataclass
class MapState:
    total_tiles: int
    revealed: int = 0
    current_position: int = 0  # 0-indexed
    cols: int = 20
    tile_log: list = field(default_factory=list)  # list of dicts: {pos, type, text}


class MapEngine:
    def __init__(self, theme, total_days=175):
        self.theme = theme
        self.cols, self.rows = theme.get_map_dimensions(total_days)
        self.total_tiles = total_days
        self.state = MapState(total_tiles=total_days, cols=self.cols)
        self._milestones = theme.get_milestone_names()
        self._init_tiles()

    def _init_tiles(self):
        tiles = []
        for i in range(self.total_tiles):
            tile = {"pos": i, "type": "fog", "text": ""}
            if (i + 1) in self._milestones:
                tile["type"] = "milestone"
                tile["text"] = self._milestones[i + 1]
            tiles.append(tile)
        self.state.tile_log = tiles

    def get_state(self):
        return {
            "total_tiles": self.state.total_tiles,
            "revealed": self.state.revealed,
            "current_position": self.state.current_position,
        }

    def reveal_tile(self, count=1):
        """Reveal `count` tiles. Returns list of newly revealed tile infos."""
        newly_revealed = []
        for _ in range(count):
            if self.state.revealed >= self.state.total_tiles:
                break
            pos = self.state.revealed
            tile = self.state.tile_log[pos]
            if tile["type"] == "milestone":
                newly_revealed.append({"pos": pos, "is_milestone": True, "name": tile["text"]})
            else:
                newly_revealed.append({"pos": pos, "is_milestone": False})
            tile["revealed"] = True
            self.state.revealed += 1
        self.state.current_position = self.state.revealed
        return newly_revealed

    def is_complete(self):
        return self.state.revealed >= self.state.total_tiles

    def is_milestone(self, day_number):
        return day_number in self._milestones

    def get_milestone_info(self, day_number):
        return self._milestones.get(day_number)

    def render(self):
        """Render the map as a Rich-markup string."""
        ts = self.theme.tile_set
        lines = []
        for row in range(self.rows):
            line = ""
            for col in range(self.cols):
                idx = row * self.cols + col
                if idx >= self.total_tiles:
                    line += " "
                    continue
                if idx < self.state.revealed:
                    tile = self.state.tile_log[idx]
                    if tile.get("type") == "milestone" and idx == self.state.current_position - 1:
                        line += f"[{ts.milestone_style}]{ts.milestone}[/]"
                    elif tile.get("type") == "milestone":
                        line += f"[{ts.milestone_style}]{ts.milestone}[/]"
                    elif idx == self.state.current_position - 1:
                        line += f"[{ts.ship_style}]{ts.ship}[/]"
                    else:
                        line += f"[{ts.explored_style}]{ts.explored_empty}[/]"
                else:
                    line += f"[{ts.unexplored_style}]{ts.unexplored}[/]"
            lines.append(line)
        return "\n".join(lines)
```

```python
"""LLM client for bestman. Handles narrative generation, coach dialogue,
and random event generation via OpenAI-compatible API."""

from openai import OpenAI


def build_voyage_log_prompt(day, stage_name, days_remaining, streak, extra_done=0, narrative_tone=""):
    extra_note = f"\nThe voyager went above and beyond today (+{extra_done} extra). Celebrate this." if extra_done > 0 else ""
    streak_note = f"\nThis is day {streak} of a streak. The crew is in good spirits." if streak > 3 else ""
    return f"""Today is Day {day} of the voyage. Current stage: {stage_name}.
{days_remaining} days until landfall.
{extra_note}{streak_note}

Write today's ship log entry. One short paragraph (3-5 sentences in Chinese).
Focus on the journey's atmosphere, the sea, the sky, the crew's mood.
{narrative_tone}"""


def build_coach_system_prompt(persona, progress):
    return f"""You are a seasoned fitness navigator aboard the ship "Bestman".
You help {persona.get('name', 'the voyager')} stay on course.

The voyager's goal: {"减肥，站在婚礼上自信地交谈，不再因身材焦虑而躲在角落。"}
Starting weight: {persona.get('start_weight_kg', '?')}kg. Height: {persona.get('height_cm', '?')}cm.
Target date: {persona.get('target_date', '2026-10-25')}.

Current progress:
- Streak: {progress.get('streak', 0)} days
- Total days completed: {progress.get('total_completed', 0)}
- Skip tokens available: {progress.get('skip_tokens', 0)}

Your role:
1. When asked, suggest today's base task. Keep it minimal — the #1 goal is completing something.
   Recommend: 死虫式 3×10, 静蹲 2×30秒, 深蹲 3×10, or 爬5层楼梯.
2. Adjust based on how the voyager feels today.
3. NEVER shame or pressure. This is a long voyage. Rest is part of it.
4. Speak like a wise navigator, not a drill sergeant. Use nautical metaphors sparingly.
5. Responses in Chinese, 2-4 sentences, warm and practical. Keep it brief."""


class LLMClient:
    def __init__(self, api_key, base_url="https://api.openai.com/v1", model="gpt-4o-mini"):
        self.client = OpenAI(api_key=api_key, base_url=base_url)
        self.model = model

    def generate(self, system, user, temperature=0.9):
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": system},
                {"role": "user", "content": user},
            ],
            temperature=temperature,
            max_tokens=300,
        )
        return response.choices[0].message.content

    def generate_voyage_log(self, day, stage_name, days_remaining, streak, narrative_tone, extra_done=0):
        prompt = build_voyage_log_prompt(
            day=day,
            stage_name=stage_name,
            days_remaining=days_remaining,
            streak=streak,
            extra_done=extra_done,
            narrative_tone=narrative_tone,
        )
        return self.generate(
            system="You are a ship logkeeper. Write in Chinese. Be vivid, concise, and encouraging.",
            user=prompt,
            temperature=0.9,
        )

    def chat_with_coach(self, persona, progress, user_message, history=None):
        system = build_coach_system_prompt(persona=persona, progress=progress)
        messages = [{"role": "system", "content": system}]
        if history:
            messages.extend(history)
        messages.append({"role": "user", "content": user_message})
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=0.7,
            max_tokens=250,
        )
        return response.choices[0].message.content

    def generate_event(self, day, stage_name, streak, narrative_tone, weather="fair"):
        prompt = f"""Day {day}, {stage_name} stage. Streak: {streak}. Weather: {weather}.
Generate a random nautical encounter for the voyage. One sentence in Chinese.
Examples: "西北方出现一群海豚随船伴游。" "风向突变，今天需要额外掌舵。"
"瞭望手报告远处有鲸鱼喷水。" "甲板上的水手们在月光下唱歌。"
Keep it brief, vivid, and fitting the sailing theme."""
        return self.generate(
            system="Generate a brief random nautical event. One sentence in Chinese.",
            user=prompt,
            temperature=1.0,
        )
```

```python
"""Core game logic for bestman: daily task completion, streak tracking,
skip tokens, stage progression, and milestone triggering."""

import datetime
from bestman.state import BestmanState
from bestman.map_engine import MapEngine
from bestman.themes import get_theme


def get_stage_for_day(day, stages):
    """Return the stage dict for a given day number (1-indexed)."""
    cumulative = 0
    for stage in stages:
        cumulative += stage["days"]
        if day <= cumulative:
            return stage
    return stages[-1]


class Voyage:
    def __init__(self, db_path=None, config=None, llm=None):
        self.state = BestmanState(db_path)
        self.config = config or {}
        self.llm = llm
        voyage_cfg = self.config.get("voyage", {})
        total_days = voyage_cfg.get("total_days", 175)
        theme_name = self.config.get("theme", "naval")
        theme = get_theme(theme_name)
        self.map_engine = MapEngine(theme=theme, total_days=total_days)
        self._restore_map_progress()

    def _restore_map_progress(self):
        progress = self.state.get_progress()
        completed = progress["total_completed"]
        if completed > 0:
            self.map_engine.reveal_tile(completed)

    def get_progress(self):
        progress = self.state.get_progress()
        voyage_cfg = self.config.get("voyage", {})
        stage_info = self.state.get_current_stage_info(
            start_date=voyage_cfg.get("start_date", "2026-05-03"),
            total_days=voyage_cfg.get("total_days", 175),
        )
        current_day = stage_info["current_day"]
        stage = get_stage_for_day(current_day, voyage_cfg.get("stages", []))
        return {
            **progress,
            **stage_info,
            "current_stage": stage["name"],
            "current_stage_desc": stage.get("description", ""),
            "theme": self.config.get("theme", "naval"),
        }

    def get_daily_task(self):
        voyage_cfg = self.config.get("voyage", {})
        return voyage_cfg.get("default_daily_task", "死虫式 3×10 + 静蹲 2×30秒")

    def complete_daily_task(self, date, task_done, extra=0):
        max_extra = self.config.get("voyage", {}).get("max_extra_per_day", 3)
        extra = min(extra, max_extra)
        self.state.record_day(date=date, completed=True, extra=extra, task_done=task_done)
        tiles = 1 + extra
        revealed = self.map_engine.reveal_tile(tiles)
        return {
            "completed": True,
            "tiles_revealed": tiles,
            "new_milestones": [r for r in revealed if r.get("is_milestone")],
            "is_complete": self.map_engine.is_complete(),
        }

    def skip_day(self, date, use_token=False):
        if not use_token:
            return {"skipped": False, "message": "Use a skip token to rest without losing streak."}
        success = self.state.use_skip_token()
        if not success:
            return {"skipped": False, "message": "No skip tokens available. Earn them with 7-day streaks."}
        self.state.record_day(date=date, completed=True, extra=0, task_done="(休整驿站)", used_skip=True)
        return {"skipped": True, "message": "Waystation rest. The crew recovers. Tomorrow we sail on."}

    def render_map(self):
        return self.map_engine.render()

    def get_map_state(self):
        return self.map_engine.get_state()

    def get_recent_logs(self, limit=5):
        return self.state.get_voyage_logs(limit=limit)

    def save_log(self, date, text, event_type="narrative"):
        self.state.save_voyage_log(date, text, event_type)
```

```python
"""CLI entry point for bestman. Parses commands and renders the TUI via Rich.

Commands:
  bestman          Show daily dashboard (map + task)
  bestman done     Mark today's task complete
  bestman done +N  Complete with N extra tiles
  bestman skip     Use a skip token for rest day
  bestman talk     Chat with the AI fitness coach
  bestman log      View recent voyage logs
"""

import re
import datetime
import sys
from rich.console import Console
from rich.panel import Panel
from rich.table import Table
from rich.text import Text
from rich.layout import Layout
from rich.live import Live

from bestman.config import load_config, load_env, get_llm_client_kwargs, get_model_name
from bestman.voyage import Voyage
from bestman.llm import LLMClient


console = Console()


def parse_command(user_input):
    """Parse a command string into (command, args_dict)."""
    s = user_input.strip().lower()
    if not s:
        return ("help", {})
    if s == "done":
        return ("done", {"extra": 0})
    m = re.match(r"done\s+\+(\d+)", s)
    if m:
        extra = int(m.group(1))
        return ("done", {"extra": min(extra, 3)})
    if s == "skip":
        return ("skip", {})
    if s == "talk":
        return ("talk", {})
    if s == "log":
        return ("log", {})
    if s in ("quit", "exit"):
        return ("quit", {})
    return ("unknown", {})


def build_dashboard(voyage, config):
    """Build the Rich layout for the daily dashboard."""
    progress = voyage.get_progress()
    theme_display = config.get("theme", "naval")

    # Map
    map_str = voyage.render_map()

    # Status line
    status = Text()
    status.append(f"\n🚢 {theme_display.upper()} VOYAGE", style="bold cyan")
    status.append(f"\nDay {progress['current_day']}/{progress['total_days']}")
    status.append(f" · Stage: {progress['current_stage']}", style="yellow")
    status.append(f"\n{progress['current_stage_desc']}", style="dim")
    status.append(f"\n\nStreak: {progress['streak']} days 🔥", style="bold green")
    status.append(f" · Skip tokens: {progress['skip_tokens']}  🛡️", style="dim")
    status.append(f"\nDays remaining: {progress['days_remaining']}")

    # Daily task
    task_text = voyage.get_daily_task()
    status.append(f"\n\n[bold]Today's Task:[/bold] {task_text}", style="bright_white")
    status.append("\n\nCommands: [done] [done +N] [skip] [talk] [log] [quit]")

    # Recent logs
    logs = voyage.get_recent_logs(3)
    log_text = Text()
    if logs:
        log_text.append("Recent Logs:", style="bold")
        for log in logs:
            log_text.append(f"\n  Day {log['date']}: {log['text'][:80]}...", style="dim italic")

    return map_str, status, log_text


def run_cli():
    load_env()
    config = load_config()

    llm_kwargs = get_llm_client_kwargs()
    model = get_model_name()
    if llm_kwargs["api_key"] and llm_kwargs["api_key"] != "sk-placeholder":
        llm = LLMClient(**llm_kwargs, model=model)
    else:
        llm = None

    voyage = Voyage(config=config, llm=llm)

    # Check if today was already completed
    today = datetime.date.today().isoformat()
    progress = voyage.get_progress()

    console.print()
    console.rule("[bold cyan]bestman[/] — 航向新大陆")
    map_str, status, log_text = build_dashboard(voyage, config)
    console.print(map_str)
    console.print(status)
    if str(log_text):
        console.print(log_text)

    # Interactive command loop
    while True:
        try:
            user_input = console.input("\n[bold green]> [/]").strip()
        except (EOFError, KeyboardInterrupt):
            console.print("\nFair winds, voyager.")
            break

        cmd, args = parse_command(user_input)

        if cmd == "done":
            extra = args.get("extra", 0)
            result = voyage.complete_daily_task(
                date=today,
                task_done=voyage.get_daily_task(),
                extra=extra,
            )
            console.print(f"\n[green]✓ Done![/] +{result['tiles_revealed']} tile(s) revealed.")

            # LLM narrative
            if llm:
                progress = voyage.get_progress()
                theme = voyage.config.get("theme", "naval")
                from bestman.themes import get_theme
                theme_obj = get_theme(theme)
                with console.status("[cyan]Writing voyage log...[/]"):
                    log_entry = llm.generate_voyage_log(
                        day=progress["current_day"],
                        stage_name=progress["current_stage"],
                        days_remaining=progress["days_remaining"],
                        streak=progress["streak"],
                        narrative_tone=theme_obj.get_narrative_tone(),
                        extra_done=extra,
                    )
                voyage.save_log(today, log_entry)
                console.print(f"\n[cyan]{log_entry}[/]")

                # Random event (20% chance)
                import random
                if random.random() < 0.2:
                    event = llm.generate_event(
                        day=progress["current_day"],
                        stage_name=progress["current_stage"],
                        streak=progress["streak"],
                        narrative_tone=theme_obj.get_narrative_tone(),
                    )
                    console.print(f"\n[yellow]⚡ {event}[/]")

            if result["new_milestones"]:
                for m in result["new_milestones"]:
                    console.print(f"\n[bold magenta]✦ MILESTONE: {m['name']} reached![/]")

            if result["is_complete"]:
                console.print("\n[bold green]🎯 LAND HO! New world reached![/]")

            # Refresh display
            map_str, status, log_text = build_dashboard(voyage, config)
            console.print()
            console.print(map_str)
            console.print(status)

        elif cmd == "skip":
            result = voyage.skip_day(date=today, use_token=True)
            if result["skipped"]:
                console.print(f"\n[blue]🛡️ {result['message']}[/]")
            else:
                console.print(f"\n[red]✗ {result['message']}[/]")

        elif cmd == "talk":
            if not llm:
                console.print("[red]LLM not configured. Set OPENAI_API_KEY.[/]")
                continue
            console.print("[cyan]Navigator: 今天感觉如何？[/]")
            while True:
                try:
                    msg = console.input("[bold yellow]You> [/]").strip()
                except (EOFError, KeyboardInterrupt):
                    break
                if msg.lower() in ("quit", "exit", "done", "back"):
                    break
                progress = voyage.get_progress()
                persona = config.get("profile", {})
                with console.status("[cyan]Navigator thinking...[/]"):
                    response = llm.chat_with_coach(persona=persona, progress=progress, user_message=msg)
                console.print(f"[cyan]Navigator: {response}[/]")

        elif cmd == "log":
            logs = voyage.get_recent_logs(10)
            if not logs:
                console.print("[dim]No voyage logs yet. Complete a day to start the chronicle.[/]")
            for log in logs:
                console.print(f"[dim]{log['date']}[/] [cyan]{log['text']}[/]")

        elif cmd == "quit":
            console.print("[cyan]Fair winds and following seas, voyager.[/]")
            break

        elif cmd == "help":
            map_str, status, log_text = build_dashboard(voyage, config)
            console.print(map_str)
            console.print(status)

        else:
            console.print("[red]Unknown command. Try: done, done +2, skip, talk, log, quit[/]")


if __name__ == "__main__":
    run_cli()
```

```
   Worktree B              Worktree C

                    ▼
              Phase 3 (Voyage + CLI)
```

**If you have 2 workers available:**
- After Phase 1 is committed, dispatch B and C in parallel worktrees.
- Merge both, then run Phase 3 in a single worktree.

**If single worker:**
- Run sequentially: Phase 1 → B → C → Phase 3 → Phase 4.
